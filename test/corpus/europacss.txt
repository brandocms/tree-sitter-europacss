Color rule
================================================================================

  @color fg primary;
}

--------------------------------------------------------------------------------

(stylesheet
  (rule_set
    (selectors
      (class_selector
        (class_name
          (identifier))))
    (block
      (europacss_color
        (europacss_keyword)
        target: (europacss_color_target)
        value: (plain_value)))))
